<script>
function captureChat() {
    // 캡처할 영역 선택
    const captureArea = document.getElementById('capture-area');
    
    // 버튼 텍스트 변경 (로딩 중 표시)
    const btn = document.querySelector('.btn-capture');
    const originalText = btn.innerText;
    btn.innerText = "⏳ 저장 중...";
    btn.disabled = true;

    // html2canvas 실행
    html2canvas(captureArea, {
        scale: 3, // 숫자가 클수록 고화질 (기본 1, 3이면 3배 해상도)
        useCORS: true, // 외부 이미지(Imgur) 로드를 위해 필수
        backgroundColor: '#333333' // 캡처 시 배경색 지정
    }).then(canvas => {
        // 이미지 다운로드 링크 생성
        const link = document.createElement('a');
        link.download = 'line_chat_capture.png'; // 저장될 파일 이름
        link.href = canvas.toDataURL('image/png'); // 캔버스를 PNG 이미지로 변환
        link.click(); // 자동 클릭하여 다운로드 실행
        
        // 버튼 원래대로 복구
        btn.innerText = originalText;
        btn.disabled = false;
    }).catch(err => {
        console.error("캡처 중 오류 발생:", err);
        alert("캡처에 실패했습니다. (GitHub Pages 같은 웹 서버 환경에서 실행해주세요!)");
        btn.innerText = originalText;
        btn.disabled = false;
    });
}
</script>
